````ts
import { readFileSync, existsSync } from 'node:fs'
import { join } from 'node:path'
import { fileURLToPath } from 'node:url'

import { describe, expect, it } from 'vitest'

const root = fileURLToPath(new URL('..', import.meta.url))
const DOCS = [
  'packages/ui-web/README.md',
  'packages/client-web/README.md',
  'packages/survey-web/README.md',
  'packages/survey-web/MIGRATION.md',
]

/** `@mbc-cqrs-serverless/<pkg>[/<entry>]` → the built module. */
function distOf(specifier: string): string {
  const [, pkg, entry] =
    specifier.match(/^@mbc-cqrs-serverless\/([a-z-]+)(?:\/(.+))?$/) ?? []
  return join(
    root,
    'packages',
    pkg ?? '',
    'dist',
    entry ? `${entry}/index.js` : 'index.js'
  )
}

function importsIn(markdown: string): { names: string[]; from: string }[] {
  const found: { names: string[]; from: string }[] = []
  for (const [, info, code] of markdown.matchAll(
    /```(\w[^\n]*)\n([\s\S]*?)```/g
  )) {
    if (!info || !/^(ts|tsx)\b/.test(info) || info.includes('v0')) continue
    for (const [, list, from] of (code ?? '').matchAll(
      /import\s*\{([^}]*)\}\s*from\s*'(@mbc-cqrs-serverless\/[^']+)'/g
    )) {
      const names = (list ?? '')
        .split(',')
        .map((name) => name.trim())
        .filter((name) => name && !name.startsWith('type '))
        .map((name) => name.split(/\s+as\s+/)[0]!.trim())
      found.push({ names, from: from ?? '' })
    }
  }
  return found
}

describe('documentation imports', () => {
  for (const doc of DOCS) {
    it(`${doc} imports only names its packages export`, async () => {
      const path = join(root, doc)
      expect(existsSync(path), doc).toBe(true)
      const imports = importsIn(readFileSync(path, 'utf8'))
      expect(imports.length, doc).toBeGreaterThan(0)
      for (const { names, from } of imports) {
        const mod = (await import(distOf(from))) as Record<string, unknown>
        for (const name of names)
          expect(name in mod, `${doc}: ${name} from ${from}`).toBe(true)
      }
    })
  }
})
````

````md
## Styles

Import one stylesheet once (if you use `survey-web`, import its `styles.css` instead — it contains every ui-web class it uses):

```ts
import '@mbc-cqrs-serverless/ui-web/styles.css'
```

- `styles.css` (default): library utilities are unlayered and its base reset and tokens sit in your `@layer base`, so it works with Tailwind v3 (3.4 or later), Tailwind v4 or no Tailwind. Override library styles with your own CSS loaded after it.
- `styles.layer.css` (Tailwind v4 hosts that want their own utilities to win): the same rules in `@layer mbc`. Import it **before** Tailwind, because a layer's position is fixed where its name first appears:

```css
@import '@mbc-cqrs-serverless/ui-web/styles.layer.css';
@import 'tailwindcss';
```

- Tailwind v3 3.0–3.3: load the library CSS after your own (their preflight ties with library classes).
- `className` accepts `mbc:`-prefixed classes that replace a conflicting library class, but only classes the library already ships exist in its CSS; use your own classes for anything else.

## Theme

Colours and radius are CSS variables on `:root`; dark values apply under `.dark` or `[data-mbc-theme="dark"]`:

  --mbc-radius: 0.5rem;
}

## Provider, locale and text

```tsx
import { MbcProvider } from '@mbc-cqrs-serverless/ui-web'

export function Providers({ children }: { children: React.ReactNode }) {
  return (
    <MbcProvider
      locale="en"
      messages={{ ui: { pagination: { next: 'Next' } } }}
    >
      {children}
    </MbcProvider>
  )
}
```

- `locale`: `'ja'` (default) or `'en'`; anything else is treated as `'ja'`.
- `messages`: partial overrides per namespace (`ui` here; `survey`, `surveyCreator`, `surveyTemplates` in survey-web).
- `dir`: `'ltr' | 'rtl'`; `portalContainer`: where dialogs, menus and popovers mount.

Add your own namespace with `defineMessages` and read it with `useMessages`:

```tsx
import { defineMessages, useMessages } from '@mbc-cqrs-serverless/ui-web'

type ShopMessages = { cart: string }
declare module '@mbc-cqrs-serverless/ui-web' {
  interface MbcMessageRegistry {
    shop: ShopMessages
  }
}
const shopMessages = defineMessages('shop', {
  ja: { cart: 'カート' },
  en: { cart: 'Cart' },
})

export function CartTitle() {
  return <h2>{useMessages(shopMessages).cart}</h2>
}
```
````

````md
  getAuthToken: async () => localStorage.getItem('idToken') ?? undefined,
  realtime: {
    url: 'https://abc.appsync-api.ap-northeast-1.amazonaws.com/graphql',
    auth: { type: 'USER_POOL' }, // or { type: 'API_KEY', apiKey: '…' }
  },
})
```

Options: `apiUrl`, `tenantCode` (string or function), `getAuthToken`, `headers`, `fetch`, `timeoutMs` (HTTP, default 30 000), `realtime` (omit it and `command()` resolves right after the HTTP response).

## HTTP

```ts
const page = await client.get<{ items: unknown[]; total: number }>(
  '/api/survey-template',
  {
    query: { page: 1, pageSize: 10, orderBys: ['createdAt:desc'] }, // arrays repeat the key
  }
)
await client.post('/api/item', { name: 'A' })
```

Every call accepts `signal` (an `AbortSignal`). Failures are classes, never logs or toasts:

```ts
import {
  MbcAbortError,
  MbcApiError,
  MbcCommandError,
  MbcTimeoutError,
} from '@mbc-cqrs-serverless/client-web'
```

- `MbcApiError` — `status` and `body`; `status` is `0` when no response arrived (network failure, or `getAuthToken` threw).
- `MbcTimeoutError`, `MbcAbortError`, `MbcRealtimeError`.
- `MbcCommandError` — the backend reported `<step>:FAILED` for a command (`requestId`, `status`).

## Commands

```ts
const { data, requestId } = await client.command<{ requestId: string }>(
  {
    method: 'PUT',
    path: '/api/survey-template/abc',
    body: { name: 'New name' },
  },
  { timeoutMs: 60_000 }
)
```

`command()` subscribes to command-status messages before sending, so a fast `finish:FINISHED` is never missed. A response without `requestId` rejects with `MbcApiError` (status 2xx); the mbc backend returns that when an update changed nothing.

## Real-time messages

```ts
const subscription = client.messages?.subscribe(
  { tenantCode: 'my-tenant', action: 'command-status' },
  (message) => console.info(message.id, message.content),
  (error) => console.error(error)
)
await subscription?.ready
subscription?.unsubscribe()
```

One WebSocket per client, opened on the first subscription and closed after the last; it keeps alive, reconnects with backoff and re-subscribes.

## React

```tsx
import {
  MbcClientProvider,
  useCommand,
  useMbcClient,
} from '@mbc-cqrs-serverless/client-web/react'

export function SaveButton({ body }: { body: unknown }) {
  const save = useCommand(({ client, signal, onPhase }, payload: unknown) =>
    client.command(
      { method: 'POST', path: '/api/item', body: payload },
      { signal, onPhase }
    )

  )
}
```
````

````md
## SurveyForm

```tsx
'use client'
import {
  SurveyForm,
  type Survey,
  type SurveyAnswers,
} from '@mbc-cqrs-serverless/survey-web'

export function Answer({
  survey,
  saved,
}: {
  survey: Survey
  saved?: SurveyAnswers
}) {
  return (
    <SurveyForm
      schema={survey}
      defaultValues={saved}
      onSubmit={async (answers, meta) => {
        await fetch('/api/answers', {
          method: 'POST',
          body: JSON.stringify({ answers, meta }),
        })
      }}
      onSubmitError={(error) => console.error(error)}
    />
  )
}
```

- Sections with Back / Next / Submit, branching by option or section action; answers in sections the respondent did not visit are submitted as `undefined`.
- `onSubmit` may return a promise: the button shows a loading state, a rejection shows a translated error and calls `onSubmitError`.
- Also: `disabled`, `submitLabel`, `children` (above the questions), `onSectionChange(sectionId)`, `className`, and `components` to replace a question renderer (`QuestionRendererProps`).
- `schema` may be stored JSON: it is parsed (defaults filled); items that fail validation render with v0's fallbacks.

'use client'

To submit answers from a `SurveyForm` and keep its error state, throw when the command fails (`run` resolves `undefined` instead of throwing):

```tsx
'use client'
import { SurveyForm, type Survey } from '@mbc-cqrs-serverless/survey-web'
import { useSubmitSurveyAnswer } from '@mbc-cqrs-serverless/survey-web/mbc'

export function Respond({
  surveyId,
  survey,
}: {
  surveyId: string
  survey: Survey
}) {
  const submit = useSubmitSurveyAnswer()
  return (
    <SurveyForm
      schema={survey}
      onSubmit={async (answer) => {
        // The command's error is in submit.error on the next render.
        if (!(await submit.run({ surveyId, answer })))
          throw new Error('The answer was not sent')
      }}
    />
  )
}
```

## Validating answers on a server (`/core`)

```ts
import {
  collectAnswers,
  buildPages,
  parseSurvey,
  resolvePath,
  validateAnswers,
} from '@mbc-cqrs-serverless/survey-web/core'

export function check(storedSurvey: unknown, answers: Record<string, unknown>) {
  const parsed = parseSurvey(storedSurvey)
  if (!parsed.success) throw parsed.error
  const errors = validateAnswers(parsed.data, answers) // {} when valid
  const pages = buildPages(parsed.data)
  const { answers: clean } = collectAnswers(
    pages,
    resolvePath(pages, answers),
    answers as never
  )
  return { errors, clean } // clean: unvisited sections' answers are undefined
}
```

`/core` imports only zod, so it loads fast and needs no React.
````

````md
## 6. Behaviour changes

Saved surveys and answers keep their shapes. What behaves differently:

**Answers**

- Answers in sections the respondent did not reach (an abandoned branch) are submitted as `undefined`.
- An untouched time or duration question is submitted as `undefined` (0.0.x sent empty parts).
- Linear scale starts at the schema's `min` (default 0); 0.0.x used `min || 1`.

**Form**

- Multiple choice keeps "Other" when a regular option is ticked.
- Rating is a keyboard-operable radio group.
- Date-time without a year writes year 0 consistently and ignores the stored year when reading.
- Placeholders and units come from the translatable `survey` messages (same Japanese text).
- Enter in a field on a non-final section goes to the next section instead of submitting.
- A second click of a double-click on Next is ignored (it could submit the next section).
- Option shuffling no longer reshuffles when the page re-renders.

**Creator**

- Changing between choice types keeps the options; switching to multiple choice drops `nextSectionId`; switching to a dropdown drops "Other"; switching from a non-choice type adds one option.
- "Submit form" after a section works.
- Reordering sections keeps questions placed before the first section header.
- Nothing can be added to a locked section, including duplicates of its questions.
- Errors are shown on the cards after a save attempt; missing branch targets show a warning.
- Option previews keep the schema order.
- A duplicated section's title gets a translated suffix.
- The preview's submit shows a note instead of `alert()`.
- Turning branching off clears every option's `nextSectionId`; turning the description off removes the key.
- Deleting or merging the selected item selects the item before it.

**Template screens**

- No routing or toasts inside the package (section 3).
- Back asks before discarding a changed new survey too (0.0.x asked only when editing).
- List state is internal unless you pass `state`.
- The list's error state has a retry button; pages are keyboard-reachable links.
- Saving unchanged data succeeds quietly (0.0.x reported a failure).
- A failed command status (`<step>:FAILED`) is reported through `onError` (0.0.x waited forever).
- Card dates follow the locale and appear after hydration.
- Ids in API paths are URL-encoded.
````
