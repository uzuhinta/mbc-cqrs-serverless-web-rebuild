> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `@mbc-cqrs-serverless/ui-web` ships the provider + typed i18n, the 20 primitives survey needs plus `Field` helpers — each a v0 look-alike that works on React 18 and 19, inside or outside a host Tailwind — and a Vite playground to see them.

**Architecture:** Primitives are v0's shadcn components re-written with `forwardRef` (React 18-safe), `mbc:`-prefixed classes, `data-mbc` on every root and portal content, and strings from the `ui` message namespace. A small context (`MbcProvider`) carries locale, message overrides, and the portal container. `Field` wires ids and `aria-*` into any control through `useFieldControl`. Look parity is pinned by tests that compare each element's classes (prefix stripped) to v0's class strings.

**Tech Stack:** SP1 toolchain + individual `@radix-ui/react-*` packages (Task 5 ruling: the umbrella `radix-ui` made `.d.ts` emit fail with TS2883), `react-day-picker` ^9.14.0, `tw-animate-css` ^1.4.0 (`/prefix` build), `axe-core` (a11y tests), Vite ^8.3.2 + `@vitejs/plugin-react` ^6.1.1 (playground).

- [ ] **Step 4: Implement `src/i18n/merge.ts`**

```ts
function isPlainObject(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value)
}

/** Deep-merges message overrides into a base tree without mutating either. */
export function mergeMessages<T>(base: T, override: unknown): T {
  if (!isPlainObject(base) || !isPlainObject(override)) return base

  const result: Record<string, unknown> = { ...base }
  for (const [key, value] of Object.entries(override)) {
    if (value === undefined) continue
    const current = result[key]
    result[key] =
      isPlainObject(current) && isPlainObject(value)
        ? mergeMessages(current, value)
        : value
  }
  return result as T
}
```

```ts
import type { UiMessages } from './ui-messages'

export type Locale = 'ja' | 'en'

/** A message is a string, or a function of named parameters returning a string. */
export type Message = string | ((params: never) => string)

export type MessageTree = { readonly [key: string]: Message | MessageTree }

export type MessageOverrides<T> = {
  [K in keyof T]?: T[K] extends string
    ? string
    : T[K] extends (params: infer P) => string
      ? (params: P) => string
      : MessageOverrides<T[K]>
}

export interface MessageCatalog<T extends MessageTree> {
  readonly namespace: string
  readonly locales: Readonly<Record<Locale, T>>
}

 * Message namespaces known to `MbcProvider`. Feature packages add theirs:

 *   declare module '@mbc-cqrs-serverless/ui-web' {

export interface MbcMessageRegistry {
  ui: UiMessages
}
```

```tsx
'use client'

import { createContext, useContext, useMemo, type ReactNode } from 'react'

import { mergeMessages } from './merge'
import type {
  Locale,
  MbcMessageRegistry,
  MessageCatalog,
  MessageOverrides,
  MessageTree,
} from './types'

export type MbcMessageOverrides = {
  [K in keyof MbcMessageRegistry]?: MessageOverrides<MbcMessageRegistry[K]>
}

type MbcContextValue = {
  locale: Locale
  messages: MbcMessageOverrides
  portalContainer: HTMLElement | null | undefined
}

const MbcContext = createContext<MbcContextValue>({
  locale: 'ja',
  messages: {},
  portalContainer: undefined,
})

export type MbcProviderProps = {
  locale?: Locale
  messages?: MbcMessageOverrides
  portalContainer?: HTMLElement | null
  children?: ReactNode
}

export function MbcProvider({
  locale,
  messages,
  portalContainer,
  children,
}: MbcProviderProps) {
  const parent = useContext(MbcContext)
  const value = useMemo<MbcContextValue>(
    () => ({
      locale: locale ?? parent.locale,
      messages: messages
        ? mergeMessages(parent.messages, messages)
        : parent.messages,
      portalContainer:
        portalContainer === undefined
          ? parent.portalContainer
          : portalContainer,
    }),
    [locale, messages, portalContainer, parent]
  )
  return <MbcContext.Provider value={value}>{children}</MbcContext.Provider>
}

export function useLocale(): Locale {
  return useContext(MbcContext).locale
}

export function usePortalContainer(): HTMLElement | undefined {
  return useContext(MbcContext).portalContainer ?? undefined
}

export function useMessages<T extends MessageTree>(
  catalog: MessageCatalog<T>
): T {
  const { locale, messages } = useContext(MbcContext)
  const override = (messages as Record<string, unknown>)[catalog.namespace]
  return useMemo(
    () => mergeMessages(catalog.locales[locale], override),
    [catalog, locale, override]
  )
}
```

```ts
import { expect } from 'vitest'

function classList(value: string): string[] {
  return value.split(/\s+/).filter(Boolean)
}

/**
 * The element's classes must all be `mbc:`-prefixed and, with the prefix
 * removed, be exactly v0's classes (order-insensitive).
 */
export function expectV0Classes(element: Element, v0: string): void {
  const actual = classList(element.getAttribute('class') ?? '')
  const unprefixed = actual.filter((name) => !name.startsWith('mbc:'))

  expect(unprefixed, 'classes without the mbc: prefix').toEqual([])
  expect(actual.map((name) => name.slice(4)).sort()).toEqual(
    classList(v0).sort()
  )
}
```

```tsx
'use client'

import {
  createContext,
  forwardRef,
  useContext,
  useEffect,
  useId,
  useMemo,
  useState,
  type AriaAttributes,
  type HTMLAttributes,
  type ReactNode,
} from 'react'

import { cn } from '../lib/cn'
import { Label, type LabelProps } from './label'

type FieldContextValue = {
  controlId: string
  labelId: string
  descriptionId: string
  errorId: string
  invalid: boolean
  hasDescription: boolean
  hasError: boolean
  setHasDescription: (value: boolean) => void
  setHasError: (value: boolean) => void
}

const FieldContext = createContext<FieldContextValue | null>(null)

export type FieldProps = HTMLAttributes<HTMLDivElement> & { invalid?: boolean }

export const Field = forwardRef<HTMLDivElement, FieldProps>(function Field(
  { className, invalid = false, ...props },
  ref
) {
  const id = useId()
  const [hasDescription, setHasDescription] = useState(false)
  const [hasError, setHasError] = useState(false)
  const value = useMemo<FieldContextValue>(
    () => ({
      controlId: `${id}-control`,
      labelId: `${id}-label`,
      descriptionId: `${id}-description`,
      errorId: `${id}-error`,
      invalid,
      hasDescription,
      hasError,
      setHasDescription,
      setHasError,
    }),
    [id, invalid, hasDescription, hasError]
  )
  return (
    <FieldContext.Provider value={value}>
      <div
        ref={ref}
        data-mbc=""
        data-slot="field"
        data-invalid={invalid || undefined}
        className={cn('mbc:grid mbc:gap-2', className)}
        {...props}
      />
    </FieldContext.Provider>
  )
})

export type FieldControlProps = {
  id?: string
  'aria-describedby'?: string
  'aria-labelledby'?: string
  'aria-invalid'?: AriaAttributes['aria-invalid']
}

export function useFieldControl<P extends FieldControlProps>(
  props: P,
  options: { group?: boolean } = {}
): P {
  const field = useContext(FieldContext)
  if (!field) return props

  const describedBy =
    [
      props['aria-describedby'],
      field.hasDescription ? field.descriptionId : undefined,
      field.hasError ? field.errorId : undefined,
    ]
      .filter(Boolean)
      .join(' ') || undefined

  return {
    ...props,
    id: props.id ?? field.controlId,
    'aria-describedby': describedBy,
    'aria-invalid': props['aria-invalid'] ?? (field.invalid || undefined),
    ...(options.group
      ? { 'aria-labelledby': props['aria-labelledby'] ?? field.labelId }
      : {}),
  }
}

export const FieldLabel = forwardRef<HTMLLabelElement, LabelProps>(
  function FieldLabel(props, ref) {
    const field = useContext(FieldContext)
    return (
      <Label
        ref={ref}
        id={field?.labelId}
        {...props}
        htmlFor={props.htmlFor ?? field?.controlId}
      />
    )
  }
)

function useRegister(
  setter: ((value: boolean) => void) | undefined,
  active: boolean
) {
  useEffect(() => {
    if (!setter || !active) return
    setter(true)
    return () => setter(false)
  }, [setter, active])
}

function hasContent(children: ReactNode): boolean {
  return (
    children !== undefined &&
    children !== null &&
    children !== false &&
    children !== ''
  )
}

export const FieldDescription = forwardRef<
  HTMLParagraphElement,
  HTMLAttributes<HTMLParagraphElement>
>(function FieldDescription({ className, ...props }, ref) {
  const field = useContext(FieldContext)
  useRegister(field?.setHasDescription, true)
  return (
    <p
      ref={ref}
      id={field?.descriptionId}
      data-slot="field-description"
      className={cn(
        'mbc:text-sm mbc:font-normal mbc:text-[#475467]',
        className
      )}
      {...props}
    />
  )
})

export const FieldError = forwardRef<
  HTMLParagraphElement,
  HTMLAttributes<HTMLParagraphElement>
>(function FieldError({ className, children, ...props }, ref) {
  const field = useContext(FieldContext)
  const visible = hasContent(children)
  useRegister(field?.setHasError, visible)
  if (!visible) return null
  return (
    <p
      ref={ref}
      id={field?.errorId}
      data-slot="field-error"
      className={cn('mbc:text-destructive mbc:text-sm', className)}
      {...props}
    >
      {children}
    </p>
  )
})
```
